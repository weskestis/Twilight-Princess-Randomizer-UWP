cmake_minimum_required(VERSION 3.20)
project(pokemon_battle_cp1 LANGUAGES CXX)

add_library(pokemon_battle_core src/battle.cpp src/session.cpp)
target_include_directories(pokemon_battle_core PUBLIC include)
target_compile_features(pokemon_battle_core PUBLIC cxx_std_20)
add_executable(pokemon_battle_demo demo/main.cpp)
target_link_libraries(pokemon_battle_demo PRIVATE pokemon_battle_core)
add_executable(pokemon_battle_tests tests/battle_tests.cpp)
target_link_libraries(pokemon_battle_tests PRIVATE pokemon_battle_core)
enable_testing()
add_test(NAME battle_core COMMAND pokemon_battle_tests)

set(POKEBATTLE_DUSK_SDK_DIR "" CACHE PATH "Optional pinned Dusklight SDK directory")
if (POKEBATTLE_DUSK_SDK_DIR)
    add_executable(pokemon_battle_ui_tests tests/ui_tests.cpp native/mod.cpp)
    target_include_directories(pokemon_battle_ui_tests SYSTEM PRIVATE "${POKEBATTLE_DUSK_SDK_DIR}/include")
    target_link_libraries(pokemon_battle_ui_tests PRIVATE pokemon_battle_core)
    add_test(NAME battle_ui COMMAND pokemon_battle_ui_tests)
endif ()

add_library(pokemon_model_runtime runtime/model.cpp)
target_compile_features(pokemon_model_runtime PUBLIC cxx_std_20)
add_executable(pokemon_model_runtime_tests tests/runtime_tests.cpp)
target_link_libraries(pokemon_model_runtime_tests PRIVATE pokemon_model_runtime)
add_test(NAME model_runtime COMMAND pokemon_model_runtime_tests)
add_library(pokemon_field_core field/party.cpp field/combat.cpp)
target_compile_features(pokemon_field_core PUBLIC cxx_std_20)
add_executable(pokemon_field_tests tests/field_tests.cpp)
target_link_libraries(pokemon_field_tests PRIVATE pokemon_field_core)
add_test(NAME field_encounters COMMAND pokemon_field_tests)
add_library(pokemon_field_animation field/animation.cpp)
target_link_libraries(pokemon_field_animation PUBLIC pokemon_field_core pokemon_model_runtime)
add_executable(pokemon_animation_tests tests/animation_tests.cpp)
target_link_libraries(pokemon_animation_tests PRIVATE pokemon_field_animation)
add_test(NAME field_animation COMMAND pokemon_animation_tests)
add_executable(pokemon_model_snapshot runtime/snapshot.cpp)
target_link_libraries(pokemon_model_snapshot PRIVATE pokemon_model_runtime)
