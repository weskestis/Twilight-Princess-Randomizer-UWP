cmake_minimum_required(VERSION 3.25)
add_mod(port_pokemon_model_lab FEATURES game
    SOURCES model_lab/mod.cpp model_lab/gx_renderer.cpp model_lab/field.cpp
    model_lab/field_effects.cpp field/party.cpp field/combat.cpp runtime/model.cpp
    MOD_JSON mod.json RES_DIR res BUNDLE)

# ModSDK.cmake supplies the common isolated SDK namespace, context and entrypoints.
# These newer services are not in .749's original service-rename loop.
target_compile_definitions(port_pokemon_model_lab PRIVATE
    DUSK_STATIC_POKEMON_MODEL_LAB=1
    svc_actor=port_pokemon_model_lab_svc_actor
    svc_file=port_pokemon_model_lab_svc_file
    svc_save=port_pokemon_model_lab_svc_save)
